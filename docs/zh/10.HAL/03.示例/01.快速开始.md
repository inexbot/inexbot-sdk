# 快速开始

```cpp
#include <hal/device_manager.h>
#include <hal/devices/digital_io.h>
#include <iostream>

int main() {
    auto& manager = hal::DeviceManager::instance();

    // 查询所有设备
    auto devices = manager.get_all_device_name_list();
    std::cout << "设备列表:" << std::endl;
    for (auto& name : devices) std::cout << "  - " << name << std::endl;

    // 获取数字 IO 设备
    if (auto dev = manager.get_device(hal::DeviceType::DIGITAL_IO_, "DIO_1")) {
        auto dio = std::dynamic_pointer_cast<hal::DigitalIO>(dev);
        dio->set_output(2, true);  // 通道3 输出高电平
        std::cout << "DIO_1 ch3 input: " << dio->get_input(3) << std::endl;
    }
    return 0;
}
```
